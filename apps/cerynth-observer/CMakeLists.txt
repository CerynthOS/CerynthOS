cmake_minimum_required(VERSION 3.21)
project(cerynth_observer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Quick QuickControls2 Network Concurrent)
qt_standard_project_setup(REQUIRES 6.6)
qt_add_executable(cerynth-observer main.cpp monitor.cpp monitor.h)
target_link_libraries(cerynth-observer PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Concurrent)
target_compile_options(cerynth-observer PRIVATE -Wall -Wextra -Wpedantic -Werror)
qt_add_qml_module(cerynth-observer URI Cerynth.Observer VERSION 1.0 QML_FILES Main.qml)
install(TARGETS cerynth-observer RUNTIME DESTINATION bin)
install(FILES io.cerynth.Observer.desktop DESTINATION share/applications)
